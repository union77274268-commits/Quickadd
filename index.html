Supabase sync

Connect your project so every saved product — and its photo link — is pushed to your own Supabase table, not just this browser.
Run this once in your Supabase SQL editor to set up the table and photo bucket:
create table if not exists products (
  id text primary key,
  barcode text,
  name text not null,
  mrp numeric default 0,
  qty integer default 0,
  image_url text,
  updated_at timestamptz default now()
);
alter table products enable row level security;
create policy "anon select" on products for select using (true);
create policy "anon insert" on products for insert with check (true);
create policy "anon update" on products for update using (true);
create policy "anon delete" on products for delete using (true);

insert into storage.buckets (id, name, public)
values ('product-images','product-images', true)
on conflict (id) do nothing;
create policy "public read images" on storage.objects
  for select using (bucket_id = 'product-images');
create policy "anon upload images" on storage.objects
  for insert with check (bucket_id = 'product-images');
Note: this opens the table and photo bucket to anyone holding your anon key, which is fine for a single-shop tool but worth knowing.
